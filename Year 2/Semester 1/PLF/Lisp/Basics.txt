;;;; EQUALITY ------------------------------------------------------------------------------------------------------------------
;; values have a ' in front.
; name = variable ; knightu_ba = value
(defparameter name 'Knightu_ba)

; eq OR equal for: numbers, strings, lists
(format t "(eq name 'Knightu_ba) = ~d ~%" (eq name 'Knightu_ba))
(format t "(equal name 'Knightu_ba) = ~d ~%" (equal name 'Knightu_ba))
(format t "~d ~%" (eq 10 11)) ; NIL
(format t "~d ~%" (equal 10 11)) ; NIL
(format t "(equal \"String\" \"string\") = ~d ~%" (equal "String" "string")) ; NIL
(format t "(equal (list 1 2 3) (list 1 2 3)) = ~d ~%" (equal (list 1 2 3) (list 1 2 3)))
