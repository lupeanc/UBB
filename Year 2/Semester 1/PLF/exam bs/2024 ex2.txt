(defun multiply (l prevNode)
    (cond
        ((and (numberp l) (numberp prevNode))
            (* l prevNode)        
        )
        ((and (numberp l) (atom prevNode))
            (* l l)
        )
        ((atom l)
            l
        )
        (t
            (mapcar #'(lambda (a) (multiply a l)) l)
        )
    )
)

absDif(A,B,R):-
    A>=B,
    R is A-B.
absDif(A,B,R):-
    B>A,
    R is B-A.

checkAbsDif([H1,H2]):-
    absDif(H1,H2,R),
    R =< 2.
checkAbsDif([H1,H2|T]):-
    absDif(H1,H2,D),
    D=<2,
    checkAbsDif([H2|T]).

oneSol(L, R):-
    perm(L,R),
    checkAbsDif(L).
