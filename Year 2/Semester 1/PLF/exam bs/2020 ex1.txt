perm([],[]).
perm([H|T],R):-
    perm(T,R1),
    insert(H,R1,R).

absDif(A,B,R):-
    A>=B,
    R is A-B.
absDif(A,B,R):-
    B>A,
    R is B-A.

checkAbsDif([H1,H2]):-
    absDif(H1,H2,R),
    R=<3.
checkAbsDif([H1,H2|T]):-
    absDif(H1,H2,R),
    R=<3,
    checkAbsDif([H2|T]).

oneSol(L,R):-
    perm(L,R),
    checkAbsDif(R).

allSols(L,R):-
    findall(Perms, oneSol(L,Perms), R).

(defun remove3 (l)
    (cond
        ((and (numberp l) (equal (mod l 3) 0))
            nil
        )
        ((atom l) (list l))
        (t
            (list (mapcan #'(lambda (a) (remove3 a)) l))
        )
    )
)

(defun main(l)
    (car (remove3 l))
)
