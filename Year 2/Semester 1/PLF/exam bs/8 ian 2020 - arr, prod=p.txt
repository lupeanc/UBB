% o ian 2020, arr, prod of el = P

insert(E,L,[E|L]).
insert(E,[H|T], [H|R]):-
    insert(E,T,R).

arr([H|_], 1, [H]).
arr([_|T], K, R):-
    arr(T,K,R).
arr([H|T], K, R):-
    K>1,
    K1 is K-1,
    arr(T,K1,R1),
    insert(H,R1,R).

prod([], 1).
prod([H|T], P):-
    prod(T,P1),
    P is H * P1.

checkProd(L, P):-
    prod(L,Prod),
    P =:= Prod.

oneSol(L, K, P, Arr):-
    arr(L,K,Arr),
    checkProd(Arr, P).

allSols(L,K, P, R):-
    findall(Arr, oneSol(L,K,P,Arr), R).
