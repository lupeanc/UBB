% B. Dându-se o listă formată din numere întregi, să se genereze ȋn PROLOG lista submulţimilor
%    cu cel puţin N elemente având suma divizibilă cu 3. 
%    Se vor scrie modelele matematice și modelele de flux pentru predicatele folosite.

% Exemplu- pentru lista L=[2,3,4] și N=1 ⇒ [[3],[2,4],[2,3,4]] (nu neapărat în această ordine)

subset([],[]).
subset([H|T],[H|R]):-
    subset(T,R).
subset([_|T], R):-
    subset(T,R).

countElems([],0).
countElems([_|T], R):-
    countElems(T,R1),
    R is 1+R1.

sum([],0).
sum([H|T], R):-
    sum(T,R1),
    R is H+R1.

checkProps(L,N):-
    countElems(L,Count),
    Count>= N,
    sum(L,S),
    S mod 3 =:= 0.

oneSol(L,N,Sub):-
    subset(L,Sub),
    checkProps(Sub, N).

allSols(L,N,R):-
    findall(Subs, oneSol(L,N,Subs), R).
    
    
    
    
    
    
; C. Se consideră o listă neliniară. Să se scrie o funcţie LISP care să aibă ca rezultat
;    lista iniţială din care au fost eliminaţi toţi atomii de pe nivelul k 
;    (nivelul superficial se consideră 1). Se va folosi o funcție MAP.

; Exemplu pentru lista (a (1 (2 b)) (c (d)))
; a) k=2 => (a ((2 b)) ((d))) 
; b) k=1 => ((1 (2 b)) (c (d))) 
; c) k=4 =>lista nu se modifică

(defun removeK(l counter k)
    (cond
        ((and (atom l) (equal counter k))
            nil
        )
        ((atom l)
            (list l)
        )
        (t
            (list (mapcan #'(lambda (a) (removeK a (+ 1 counter) k) ) l))
        )
    )
)

(defun main(l k)
    (car (removeK l 0 k))
)

(format t "~s" (main '(a (1 (2 b)) (c (d))) 2))
