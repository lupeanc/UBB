 removeOcc([],_,[]).
removeOcc([H|T],E,[H|R]):-
    H =\=E,
    removeOcc(T,E,R).
removeOcc([H|T],E,R):-
    H=:=E,
    removeOcc(T,E,R).

    findMin(T,H,R).

% a. Determine the position of the maximal element of a linear list.
findMax([],CurrentMax,CurrentMax).
findMax([H|T], CurrentMax, Max):-
    H<CurrentMax,
    findMax(T,CurrentMax,Max).
findMax([H|T], CurrentMax, Max):-
    H>=CurrentMax,
    findMax(T,H,Max).

getPos([],_,_,[]).
getPos([H|T],CurrentPos,Max,[CurrentPos|R]):-
    H=:=Max,
    NewCurrentPos is CurrentPos+1,
    getPos(T,NewCurrentPos, Max, R).
getPos([H|T],CurrentPos,Max,R):-
    H=\=Max,
    NewCurrentPos is CurrentPos+1,
    getPos(T,NewCurrentPos, Max, R).

mainMax(L,R):-
    findMax(L,0,Max),
    getPos(L,1,Max,R).
