% 1-st, 3-rd, 7-th, 15-th element a given value e.

addElem([],_,_,[]).

    \+ (Ct=:=1; Ct=:=3; Ct=:=7; Ct=:=15),

    !,

% b. For a heterogeneous list, formed from integer numbers and list of numbers, 
% define a predicate to determine the maximum number of the list, 
% and then to replace this value in sublists with the maximum value of sublist.

getNumbers([],[]).
getNumbers([H|T],[H|R]):-
    number(H),
    getNumbers(T,R).
getNumbers([H|T],R):-
    is_list(H),
    getNumbers(T,R).

    H<CurrentMax,
    findMax(T,CurrentMax,R).
findMax([H|T],CurrentMax,R):-
    H>=CurrentMax,
    findMax(T,H,R).

    is_list(H),

    replace(H,Max,SubMax,NewSub),
