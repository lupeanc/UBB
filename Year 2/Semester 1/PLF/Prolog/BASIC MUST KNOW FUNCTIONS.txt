% GCD
gcd(0, B, B):- !.
gcd(A, 0, A):- !.
gcd(A, B, R):-
    A >= B, 
	A1 is A mod B,
	gcd(A1, B, R), !.
gcd(A, B, R):-
	A < B,
	B1 is B mod A,
	gcd(A, B1, R), !.

%LCM
lcm(A, B, R):-
    gcd(A, B, RGCD),
    R is A * B / RGCD.


% INSERT AN ELEM ON EVERY POSITION OF A LIST
% insert(elem, l1l2...ln) = 
% = {elem} U l1l2...ln
% = {l1} U insert(l2...ln, elem)

% insert(L:list, E: element, R: result list)
% (i,i,o)

insert(E,L,[E|L]).
insert(E,[H|T],[H|R]):-
    insert(E,T,R).
