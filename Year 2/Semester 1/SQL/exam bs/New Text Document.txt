)

	id int primary key identity,
	tid int references traintype(id),
	name varchar(10)
)

create table station(
	id int primary key identity,
	name varchar(100)
)


create table route(
	id int primary key identity,
	tid int references train(id),
	name varchar(100),
	
)

-- list => tabel de legatura
create table routestations(
sid int references station(id),
	rid int references route(id),
	a time,
	d time,
	primary key(sid, rid)
)

create or alter procedure addupdateroute(@rname varchar(10), @sname varchar(10), @a time, @d time)
as
	begin
		declare @sid int = (select id from station where name = @sname)
		declare @rid int = (select id from route where name = @rname)

		if not exists (select * from route where name = @rname)
		begin
			raiserror('no such route', 16,1)
			return -1
		end

		if not exists (select * from station where name = @sname)
		begin
			raiserror('no such station', 16,1)
			return -1
		end

		if exists (select * from routestations rs where rs.rid = @rid and rs.sid = @sid)
		begin
			update routestations
			set a=@a, d=@d
			where sid=@sid and rid=@rid
		end

		if not exists (select * from routestations rs where rs.rid = @rid and rs.sid = @sid)
		begin
			insert into routestations(rid,sid,a,d) values (@rid, @sid, @a, @d)
		end
	end

go

create or alter view showroutes
as
	select r.name as name
	from route r inner join routestations rs on r.id = rs.rid
	--where rs.rid=r.id
	group by r.name
	having count(rs.sid) = (select distinct count(id) from station)
go

select * from showroutes
